59/// Runs an HTTP server. Hides the codec type, like `AsyncAcceptor`
60/// does for the stream transports.
61#[async_trait]
62pub(crate) trait HttpAcceptor: Send + Sync {
63    /// Run the accept loop.
64    async fn accept_loop(&self, server: Arc<Server>) -> Result<()>;
65    fn local_endpoint(&self) -> Result<Endpoint>;
66}
67
68/// HTTP server (TCP for h1/h2, optional QUIC for h3).
69pub(crate) struct HttpServer<C> {
70    tcp_listener: TcpListener,
71    /// Encodes and decodes all HTTP bodies and HTTP/3 streams.
72    codec: C,
73    #[cfg(feature = "http3")]
74    quic_endpoint: Option<karyon_net::quic::QuicEndpoint>,
75}
76
77impl<C: JsonRpcCodec> HttpServer<C> {
78    /// Create HTTP/1.1 + HTTP/2 server.
79    pub(crate) async fn new(endpoint: &Endpoint, codec: C) -> Result<Self> {
80        let addr = SocketAddr::try_from(endpoint.clone())?;
81        let tcp_listener = TcpListener::bind(addr).await?;
82        Ok(Self {
83            tcp_listener,
84            codec,
85            #[cfg(feature = "http3")]
86            quic_endpoint: None,
87        })
88    }
89
90    /// Create HTTP/1.1 + HTTP/2 + HTTP/3 server.
91    #[cfg(feature = "http3")]
92    pub(crate) async fn new_h3(
93        endpoint: &Endpoint,
94        quic_config: karyon_net::quic::ServerQuicConfig,
95        codec: C,
96    ) -> Result<Self> {
97        let addr = SocketAddr::try_from(endpoint.clone())?;
98        let tcp_listener = TcpListener::bind(addr).await?;
99        let actual_addr = tcp_listener.local_addr()?;
100        let quic_ep_addr = Endpoint::new_quic_addr(actual_addr);
101        let quic_endpoint =
102            karyon_net::quic::QuicEndpoint::listen(&quic_ep_addr, quic_config).await?;
103        Ok(Self {
104            tcp_listener,
105            codec,
106            quic_endpoint: Some(quic_endpoint),
107        })
108    }
109}
110
111#[async_trait]
112impl<C: JsonRpcCodec> HttpAcceptor for HttpServer<C> {
113    async fn accept_loop(&self, server: Arc<Server>) -> Result<()> {
114        let codec = &self.codec;
115        // Back off on accept errors, so the loop does not spin.
116        let tcp_backoff = accept_backoff();
117
118        #[cfg(feature = "http3")]
119        if let Some(ref quic_ep) = self.quic_endpoint {
120            let quic_backoff = accept_backoff();
121            loop {
122                select(
123                    h1h2::accept_tcp(&server, &self.tcp_listener, codec, &tcp_backoff),
124                    h3::accept_h3(&server, quic_ep, codec, &quic_backoff),
125                )
126                .await;
127            }
128        }
129
130        loop {
131            h1h2::accept_tcp(&server, &self.tcp_listener, codec, &tcp_backoff).await;
132        }
133    }
134
135    fn local_endpoint(&self) -> Result<Endpoint> {
136        let addr = self.tcp_listener.local_addr()?;
137        format!("http://{addr}/")
138            .parse()
139            .map_err(|e: karyon_net::Error| Error::HttpError(e.to_string()))
140    }
141}

145/// Response with one of the pre-encoded JSON error bodies (`ERR_*`).
146pub(super) fn json_response(status: StatusCode, body: &str) -> Response<Full<Bytes>> {
147    json_response_bytes(status, DEFAULT_CONTENT_TYPE, Bytes::from(body.to_string()))
148}
149
150/// Response with a codec-encoded body.
151pub(super) fn json_response_bytes(
152    status: StatusCode,
153    content_type: &str,
154    body: Bytes,
155) -> Response<Full<Bytes>> {
156    Response::builder()
157        .status(status)
158        .header(CONTENT_TYPE, content_type)
159        .body(Full::new(body))
160        .unwrap()
161}
162
163/// Encode a JSON-RPC response with the server codec.
164pub(super) fn encode_response<C: JsonRpcCodec>(
165    codec: &C,
166    resp: &message::Response,
167) -> Result<Bytes> {
168    encode_msg(codec, &to_value(resp)?)
169}
170
171/// Build the JSON-RPC parse error response.
172pub(super) fn parse_error_response() -> message::Response {
173    message::Response {
174        error: Some(message::Error {
175            code: message::PARSE_ERROR_CODE,
176            message: FAILED_TO_PARSE_ERROR_MSG.to_string(),
177            data: None,
178        }),
179        ..Default::default()
180    }
