27/// Runs a QUIC server. Hides the codec type, like `HttpAcceptor` does
28/// for HTTP.
29#[async_trait]
30pub(crate) trait QuicAcceptor: Send + Sync {
31    /// Run the accept loop.
32    async fn accept_loop(&self, server: Arc<Server>) -> Result<()>;
33    fn local_endpoint(&self) -> Result<Endpoint>;
34}
35
36/// `Arc` reference to a `QuicAcceptor`.
37pub(crate) type QuicAcceptorRef = Arc<dyn QuicAcceptor>;
38
39/// QUIC server. Every stream is framed with `codec`.
40pub(crate) struct QuicServer<C> {
41    pub(super) endpoint: QuicEndpoint,
42    pub(super) codec: C,
43}
44
45#[async_trait]
46impl<C: JsonRpcCodec> QuicAcceptor for QuicServer<C> {
47    async fn accept_loop(&self, server: Arc<Server>) -> Result<()> {
48        // Back off on accept errors, so the loop does not spin.
49        let backoff = accept_backoff(server.task_group.executor());
50        loop {
51            match self.endpoint.accept_incoming().await {
52                Ok(incoming) => {
53                    backoff.reset();
54                    handle_quic_incoming(&server, incoming, self.codec.clone());
55                }
56                Err(err) => {
57                    error!("Accept QUIC conn: {err}");
58                    backoff.sleep().await;
59                }
60            }
61        }
62    }
63
64    fn local_endpoint(&self) -> Result<Endpoint> {
65        self.endpoint.local_endpoint().map_err(Error::from)
66    }
67}
68
69/// Run the handshake off the accept loop, then serve the connection.
70fn handle_quic_incoming<C: JsonRpcCodec>(server: &ServerRef, incoming: QuicIncoming, codec: C) {
71    let peer = incoming.peer_endpoint();
72    let srv = server.clone();
73    server.task_group.spawn(async move {
74        // Bounded by the QUIC idle timeout (QuicConfig::idle_timeout).
75        match incoming.handshake().await {
76            Ok(quic_conn) => handle_quic_conn(&srv, quic_conn, codec),
77            Err(err) => debug!("QUIC handshake with {peer} failed: {err}"),
78        }
79    });
80}

82/// Serve a QUIC connection; each incoming stream is handled
83/// as an independent request.
84fn handle_quic_conn<C: JsonRpcCodec>(server: &ServerRef, quic_conn: QuicConn, codec: C) {
85    let peer = quic_conn.peer_endpoint().ok();
86    debug!("Handle QUIC connection {peer:?}");
87
88    server.task_group.spawn_then(
89        quic_accept_streams_task(server.clone(), Arc::new(quic_conn), codec),
90        move |result: TaskResult<Result<()>>| async move {
91            if let TaskResult::Completed(Err(err)) = result {
92                debug!("QUIC conn {peer:?} dropped: {err}");
93            } else {
94                debug!("QUIC conn {peer:?} dropped");
95            }
96        },
97    );
98}

100async fn quic_accept_streams_task<C: JsonRpcCodec>(
101    server: ServerRef,
102    quic_conn: QuicConnRef,
103    codec: C,
104) -> Result<()> {
105    let peer = quic_conn.peer_endpoint().ok();
106    loop {
107        let stream = quic_conn.accept_stream().await?;
108        let conn = framed(stream, codec.clone());
109        server
110            .task_group
111            .spawn(quic_handle_stream_task(server.clone(), conn, peer.clone()));
112    }
113}
114
115async fn quic_handle_stream_task<C: JsonRpcCodec>(
116    server: ServerRef,
117    conn: FramedConn<C>,
118    peer: Option<Endpoint>,
119) -> Result<()> {
120    if let Err(err) = handle_quic_stream(server, conn, peer).await {
121        error!("Handle QUIC stream: {err}");
122    }
123    Ok(())
124}

126/// Handle a single QUIC stream: one request, one response.
127/// Upgrades to pubsub notification streaming if the method matches.
128async fn handle_quic_stream<C: JsonRpcCodec>(
129    server: ServerRef,
130    mut conn: FramedConn<C>,
131    peer: Option<Endpoint>,
132) -> Result<()> {
133    // A stream that never sends its request is dropped after
134    // `read_timeout`. Only this first read: after it, a pubsub stream
135    // waits for its unsubscribe, which is idle by design.
136    let msg = match server.config.read_timeout {
137        Some(t) => {
138            server
139                .task_group
140                .executor()
141                .timeout(t, conn.recv_msg())
142                .await??
143        }
144        None => conn.recv_msg().await?,
145    };
146
147    let req = match sanity_check(&server, msg) {
148        SanityCheckResult::NewReq(req) => req,
149        SanityCheckResult::ErrRes(res) => {
150            conn.send_msg(serde_json::json!(res)).await?;
151            return Ok(());
152        }
153    };
154
155    // Pubsub handlers need a dedicated stream for streaming notifications.
156    let handler = server.resolve_handler(&req, true);
157    if let Handler::Pubsub(_) = handler {
158        return handle_quic_subscription(&server, conn, handler, req.msg, peer).await;
159    }
160
161    let response = run_handler(handler, req.msg, None).await;
162    debug!("--> {response}");
163    conn.send_msg(serde_json::json!(response)).await?;
164    Ok(())
165}

167/// Pubsub over QUIC: split the stream so notifications stream
168/// from the writer while the reader waits for an unsubscribe.
169async fn handle_quic_subscription<C: JsonRpcCodec>(
170    server: &Server,
171    mut conn: FramedConn<C>,
172    handler: Handler<'_>,
173    req: message::Request,
174    peer: Option<Endpoint>,
175) -> Result<()> {
176    let (ch_tx, ch_rx) = async_channel::bounded(CHANNEL_SUBSCRIPTION_BUFFER_SIZE);
177    let channel = Channel::new(ch_tx, peer);
178
179    let response = run_handler(handler, req, Some(channel.clone())).await;
180    if response.error.is_some() {
181        conn.send_msg(serde_json::json!(response)).await?;
182        return Ok(());
183    }
184
185    debug!("--> {response}");
186
187    let (mut reader, mut writer) = conn.split();
188    writer.send_msg(serde_json::json!(response)).await?;
189
190    let notification_encoder = server.config.notification_encoder;
191
192    loop {
193        match select(ch_rx.recv(), reader.recv_msg()).await {
194            Either::Left(nt) => {
195                let nt = nt?;
196                let notification = notification_encoder(nt);
197                debug!("--> {notification}");
198                writer.send_msg(serde_json::json!(notification)).await?;
199            }
200            Either::Right(msg) => match msg {
201                Ok(msg) => {
202                    let response = server.handle_request(Some(channel.clone()), msg).await;
203                    debug!("--> {response}");
204                    writer.send_msg(serde_json::json!(response)).await?;
205                    channel.close();
206                    return Ok(());
207                }
208                Err(_) => {
209                    channel.close();
210                    return Ok(());
211                }
212            },
213        }
214    }
