6pub const ATTR_MAPPED_ADDRESS: u16 = 0x0001;
7pub const ATTR_ERROR_CODE: u16 = 0x0009;
8pub const ATTR_XOR_MAPPED_ADDRESS: u16 = 0x0020;
9pub const ATTR_SOFTWARE: u16 = 0x8022;

11const FAMILY_IPV4: u8 = 0x01;
12const FAMILY_IPV6: u8 = 0x02;

14/// ERROR-CODE class bits.
15const CLASS_MASK: u8 = 0x07;

17/// STUN attribute (RFC 5389 section 15). Unknown attributes are kept
18/// as raw bytes and ignored.
19#[derive(Clone, Debug, PartialEq, Eq)]
20pub enum Attribute {
21    MappedAddress(SocketAddr),
22    XorMappedAddress(SocketAddr),
23    ErrorCode { code: u16, reason: String },
24    Software(String),
25    Unknown { attr_type: u16, value: Vec<u8> },
26}
27
28impl Attribute {
29    /// Encode as TLV (type, length, value, zero padding to 4 bytes).
30    pub fn encode(&self, txn_id: &TransactionId, dst: &mut Vec<u8>) -> Result<()> {
31        let (attr_type, value) = match self {
32            Attribute::MappedAddress(sa) => (ATTR_MAPPED_ADDRESS, encode_address(sa)),
33            Attribute::XorMappedAddress(sa) => (
34                ATTR_XOR_MAPPED_ADDRESS,
35                encode_address(&xor_address(sa, txn_id)),
36            ),
37            Attribute::ErrorCode { code, reason } => (ATTR_ERROR_CODE, encode_error(*code, reason)),
38            Attribute::Software(s) => (ATTR_SOFTWARE, s.as_bytes().to_vec()),
39            Attribute::Unknown { attr_type, value } => (*attr_type, value.clone()),
40        };
41
42        if value.len() > u16::MAX as usize {
43            return Err(Error::InvalidMsg("attribute value too large".into()));
44        }
45
46        dst.extend_from_slice(&attr_type.to_be_bytes());
47        dst.extend_from_slice(&(value.len() as u16).to_be_bytes());
48        dst.extend_from_slice(&value);
49        // Pad to a 4-byte boundary.
50        for _ in 0..(4 - value.len() % 4) % 4 {
51            dst.push(0);
52        }
53        Ok(())
54    }
55
56    /// Decode one attribute. Returns (consumed bytes incl. padding, attr),
57    /// or None if `src` does not hold a complete attribute.
58    pub fn decode(txn_id: &TransactionId, src: &[u8]) -> Result<Option<(usize, Self)>> {
59        if src.len() < 4 {
60            return Ok(None);
61        }
62        let attr_type = u16::from_be_bytes([src[0], src[1]]);
63        let len = u16::from_be_bytes([src[2], src[3]]) as usize;
64        let padded = (len + 3) & !3;
65        if src.len() < 4 + padded {
66            return Ok(None);
67        }
68        let value = &src[4..4 + len];
69
70        let attr = match attr_type {
71            ATTR_MAPPED_ADDRESS => Attribute::MappedAddress(decode_address(value)?),
72            // XOR is an involution: applying it to the wire form
73            // recovers the real address.
74            ATTR_XOR_MAPPED_ADDRESS => {
75                Attribute::XorMappedAddress(xor_address(&decode_address(value)?, txn_id))
76            }
77            ATTR_ERROR_CODE => decode_error(value)?,
78            ATTR_SOFTWARE => Attribute::Software(String::from_utf8_lossy(value).into_owned()),
79            t => Attribute::Unknown {
80                attr_type: t,
81                value: value.to_vec(),
82            },
83        };
84        Ok(Some((4 + padded, attr)))
85    }
86}

88/// XOR an address with the magic cookie (and transaction id for IPv6),
89/// per RFC 5389 section 15.2. Involution: applying twice is identity.
90fn xor_address(sa: &SocketAddr, txn_id: &TransactionId) -> SocketAddr {
91    let port = sa.port() ^ (MAGIC_COOKIE >> 16) as u16;
92    let ip = match sa.ip() {
93        IpAddr::V4(v4) => {
94            let mut o = v4.octets();
95            for (b, k) in o.iter_mut().zip(MAGIC_COOKIE.to_be_bytes()) {
96                *b ^= k;
97            }
98            IpAddr::V4(Ipv4Addr::from(o))
99        }
100        IpAddr::V6(v6) => {
101            let mut key = [0u8; 16];
102            key[..4].copy_from_slice(&MAGIC_COOKIE.to_be_bytes());
103            key[4..].copy_from_slice(&txn_id.0);
104            let mut o = v6.octets();
105            for (b, k) in o.iter_mut().zip(key) {
106                *b ^= k;
107            }
108            IpAddr::V6(Ipv6Addr::from(o))
109        }
110    };
111    SocketAddr::new(ip, port)
112}

114/// Address value: [0, family, port (2 bytes), address (4 or 16 bytes)].
115fn encode_address(sa: &SocketAddr) -> Vec<u8> {
116    let mut v = Vec::with_capacity(20);
117    v.push(0);
118    match sa.ip() {
119        IpAddr::V4(ip) => {
120            v.push(FAMILY_IPV4);
121            v.extend_from_slice(&sa.port().to_be_bytes());
122            v.extend_from_slice(&ip.octets());
123        }
124        IpAddr::V6(ip) => {
125            v.push(FAMILY_IPV6);
126            v.extend_from_slice(&sa.port().to_be_bytes());
127            v.extend_from_slice(&ip.octets());
128        }
129    }
130    v
131}

133fn decode_address(value: &[u8]) -> Result<SocketAddr> {
134    if value.len() < 4 {
135        return Err(Error::InvalidMsg("address attribute too short".into()));
136    }
137    let port = u16::from_be_bytes([value[2], value[3]]);
138    let ip = match value[1] {
139        FAMILY_IPV4 if value.len() == 8 => {
140            let o: [u8; 4] = value[4..8].try_into().expect("length checked");
141            IpAddr::V4(Ipv4Addr::from(o))
142        }
143        FAMILY_IPV6 if value.len() == 20 => {
144            let o: [u8; 16] = value[4..20].try_into().expect("length checked");
145            IpAddr::V6(Ipv6Addr::from(o))
146        }
147        _ => return Err(Error::InvalidMsg("bad address family or length".into())),
148    };
149    Ok(SocketAddr::new(ip, port))
150}

152/// Error value: [0, 0, class, number] + utf8 reason. Class is the
153/// low 3 bits of its byte; the rest are reserved (RFC 5389 15.6).
154fn encode_error(code: u16, reason: &str) -> Vec<u8> {
155    let mut v = Vec::with_capacity(4 + reason.len());
156    let class = (code / 100) as u8 & CLASS_MASK;
157    v.extend_from_slice(&[0, 0, class, (code % 100) as u8]);
158    v.extend_from_slice(reason.as_bytes());
159    v
160}
161
162fn decode_error(value: &[u8]) -> Result<Attribute> {
163    if value.len() < 4 {
164        return Err(Error::InvalidMsg("error attribute too short".into()));
165    }
166    let code = (value[2] & CLASS_MASK) as u16 * 100 + value[3] as u16;
167    let reason = String::from_utf8_lossy(&value[4..]).into_owned();
168    Ok(Attribute::ErrorCode { code, reason })
169}

180    fn xor_is_involution() {
181        let addrs = ["192.168.1.1:8080", "[2001:db8::1]:3478"];
182        for a in addrs {
183            let sa: SocketAddr = a.parse().unwrap();
184            assert_eq!(xor_address(&xor_address(&sa, &txn()), &txn()), sa);
185        }
186    }

190        let sa: SocketAddr = "127.0.0.1:8080".parse().unwrap();
191        let x = xor_address(&sa, &txn());
192        assert_eq!(x.port(), 8080 ^ 0x2112);
193    }

195    #[test]
196    fn error_code_ignores_reserved_bits() {
197        // Class byte 0x84: reserved bits set, class 4.
198        let value = [0, 0, 0x84, 0, b'x'];
199        let attr = decode_error(&value).unwrap();
200        assert!(matches!(attr, Attribute::ErrorCode { code: 400, .. }));
201    }

203    #[test]
204    fn attribute_roundtrip() {
205        let attrs = vec![
206            Attribute::MappedAddress("10.0.0.1:1234".parse().unwrap()),
207            Attribute::XorMappedAddress("[::1]:3478".parse().unwrap()),
208            Attribute::ErrorCode {
209                code: 400,
210                reason: "Bad Request".into(),
211            },
212            Attribute::Software("karyon".into()),
213        ];
214        for attr in attrs {
215            let mut buf = Vec::new();
216            attr.encode(&txn(), &mut buf).unwrap();
217            assert_eq!(buf.len() % 4, 0);
218            let (n, decoded) = Attribute::decode(&txn(), &buf).unwrap().unwrap();
219            assert_eq!(n, buf.len());
220            assert_eq!(decoded, attr);
221        }
222    }
