1use std::io::{Error as IoError, ErrorKind};
2use std::result::Result as StdResult;
3
4use thiserror::Error as ThisError;
5
6use karyon_net::Error as NetError;
7
8pub type Result<T> = StdResult<T, Error>;
9
10#[derive(ThisError, Debug)]
11pub enum Error {
12    #[error(transparent)]
13    IO(#[from] IoError),
14
15    #[error(transparent)]
16    KaryonNet(#[from] NetError),
17
18    #[error("Invalid Message: {0}")]
19    InvalidMsg(String),
20
21    #[error("Timeout")]
22    Timeout,
23
24    #[error("Transaction ID Mismatch")]
25    TransactionMismatch,
26
27    #[error("No Mapped Address In Response")]
28    NoMappedAddress,
29
30    #[error("STUN Error Response: {code} {reason}")]
31    ErrorResponse { code: u16, reason: String },
32
33    #[error("Unsupported Endpoint: {0}")]
34    UnsupportedEndpoint(String),
35
36    #[error("Missing {0} Config")]
37    MissingConfig(&'static str),
38}
39
40// Required so `StunCodec` plugs into karyon_net's framing.
41impl From<Error> for NetError {
42    fn from(err: Error) -> NetError {
43        match err {
44            Error::KaryonNet(e) => e,
45            Error::IO(e) => NetError::IO(e),
46            other => NetError::IO(IoError::new(ErrorKind::InvalidData, other.to_string())),
47        }
48    }
49}
