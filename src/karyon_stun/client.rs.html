43/// UDP retransmit schedule in milliseconds (RFC 5389 7.2.1, shortened).
44#[cfg(feature = "udp")]
45const UDP_RETRIES_MS: [u64; 3] = [500, 1000, 2000];

47/// Max time for a whole query over a stream transport, including the

49#[cfg(any(feature = "tcp", feature = "quic"))]

52/// STUN binding client. Asks a STUN server for this host's public
53/// address as seen from outside the NAT.
54///
55/// The server endpoint must use an IP address (`udp://`, `tcp://`,
56/// `tls://` or `quic://`); hostname resolution is up to the caller.
57/// TLS and QUIC need a client config, see `with_tls` and `with_quic`.
58pub struct StunClient {
59    server: Endpoint,
60    #[cfg(feature = "tls")]
61    tls: Option<ClientTlsConfig>,
62    #[cfg(feature = "quic")]
63    quic: Option<ClientQuicConfig>,
64}

71            tls: None,
72            #[cfg(feature = "quic")]
73            quic: None,
74        }
75    }

92    /// Send a Binding Request and return our public address.
93    pub async fn discover(&self) -> Result<SocketAddr> {
94        match &self.server {
95            #[cfg(feature = "udp")]
96            Endpoint::Udp(..) => self.discover_udp(None).await,
97            #[cfg(feature = "tcp")]
98            Endpoint::Tcp(..) => with_timeout(self.discover_tcp()).await,
99            #[cfg(feature = "tls")]
100            Endpoint::Tls(..) => with_timeout(self.discover_tls()).await,
101            #[cfg(feature = "quic")]
102            Endpoint::Quic(..) => with_timeout(self.discover_quic()).await,
103            ep => Err(Error::UnsupportedEndpoint(ep.to_string())),
104        }
105    }
106
107    /// Like `discover`, but bind the query socket to `local`, to learn
108    /// the public mapping of that specific socket. UDP only.
109    #[cfg(feature = "udp")]
110    pub async fn discover_from(&self, local: SocketAddr) -> Result<SocketAddr> {
111        self.discover_udp(Some(local)).await
112    }

114    #[cfg(feature = "udp")]
115    async fn discover_udp(&self, local: Option<SocketAddr>) -> Result<SocketAddr> {
116        let server = SocketAddr::try_from(self.server.clone())?;
117        let local = local.unwrap_or_else(|| match server {
118            SocketAddr::V4(_) => "0.0.0.0:0".parse().expect("valid address"),
119            SocketAddr::V6(_) => "[::]:0".parse().expect("valid address"),
120        });
121
122        let conn = listen(&Endpoint::new_udp_addr(local), UdpConfig::default()).await?;
123        let request = StunMessage::binding_request();
124        let encoded = request.encode()?;
125
126        for ms in UDP_RETRIES_MS {
127            conn.send_to(&encoded, server).await?;
128            let window = Duration::from_millis(ms);
129            if let Some(res) = await_udp_response(&conn, server, &request, window).await {
130                return res;
131            }
132            debug!("STUN: no response within {ms}ms, retrying");
133        }
134        Err(Error::Timeout)
135    }
136
137    #[cfg(feature = "tcp")]
138    async fn discover_tcp(&self) -> Result<SocketAddr> {
139        let stream = connect(&self.server, Default::default()).await?;
140        query_stream(stream).await
141    }
142
143    #[cfg(feature = "tls")]
144    async fn discover_tls(&self) -> Result<SocketAddr> {
145        let config = self.tls.clone().ok_or(Error::MissingConfig("tls"))?;
146        let stream = connect(&self.server, Default::default()).await?;
147        let stream = ClientLayer::handshake(&TlsLayer::client(config), stream).await?;
148        query_stream(stream).await
149    }
150
151    #[cfg(feature = "quic")]
152    async fn discover_quic(&self) -> Result<SocketAddr> {
153        let config = self.quic.clone().ok_or(Error::MissingConfig("quic"))?;
154        let conn = QuicEndpoint::dial(&self.server, config).await?;
155        let stream = conn.open_stream().await?;
156        let res = query_stream(stream).await;
157        conn.close(0, b"done");
158        res
159    }
160}

162/// Wait up to `window` for the response to `request` from `server`.
163/// Stray, garbage, and mismatched datagrams are skipped without ending
164/// the window. Returns None when the window ends.
165#[cfg(feature = "udp")]
166async fn await_udp_response(
167    conn: &UdpConn,
168    server: SocketAddr,
169    request: &StunMessage,
170    window: Duration,
171) -> Option<Result<SocketAddr>> {
172    let deadline = Instant::now() + window;
173    let mut buf = [0u8; 1500];
174    loop {
175        let left = deadline.saturating_duration_since(Instant::now());
176        let (n, from) = match timeout(left, conn.recv_from(&mut buf)).await {
177            Ok(Ok(received)) => received,
178            Ok(Err(e)) => return Some(Err(e.into())),
179            Err(_) => return None,
180        };
181        if from != server {
182            continue;
183        }
184        let msg = match StunMessage::decode(&buf[..n]) {
185            Ok(Some((_, msg))) => msg,
186            _ => continue,
187        };
188        match validate_response(request, msg) {
189            Err(Error::TransactionMismatch) => continue,
190            res => return Some(res),
191        }
192    }
193}

195/// Bound a stream query by `STREAM_TIMEOUT_SECS`.
196#[cfg(any(feature = "tcp", feature = "quic"))]
197async fn with_timeout<F>(fut: F) -> Result<SocketAddr>
198where
199    F: Future<Output = Result<SocketAddr>>,
200{
201    let d = Duration::from_secs(STREAM_TIMEOUT_SECS);
202    timeout(d, fut).await.map_err(|_| Error::Timeout)?
203}

205/// Send one Binding Request on a byte stream and read the response.
206#[cfg(any(feature = "tcp", feature = "quic"))]
207async fn query_stream(stream: Box<dyn ByteStream>) -> Result<SocketAddr> {
208    let mut conn = framed(stream, StunCodec);
209    let request = StunMessage::binding_request();
210    conn.send_msg(request.clone()).await?;
211    let msg = conn.recv_msg().await?;
212    validate_response(&request, msg)
213}
214
215/// Check a response against its request and extract the mapped address.
216fn validate_response(request: &StunMessage, response: StunMessage) -> Result<SocketAddr> {
217    if response.txn_id != request.txn_id {
218        return Err(Error::TransactionMismatch);
219    }
220    match response.msg_type {
221        MessageType::BindingResponse => response.mapped_address(),
222        MessageType::BindingError => {
223            for attr in &response.attrs {
224                if let Attribute::ErrorCode { code, reason } = attr {
225                    return Err(Error::ErrorResponse {
226                        code: *code,
227                        reason: reason.clone(),
228                    });
229                }
230            }
231            Err(Error::ErrorResponse {
232                code: 0,
233                reason: "error response without ERROR-CODE".into(),
234            })
235        }
236        MessageType::BindingRequest => Err(Error::InvalidMsg("request sent as response".into())),
237    }
