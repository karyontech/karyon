1//! The peer id in a karyon certificate: the node's key signs the
2//! certificate's key, in a custom extension. Used by the built-in TLS and
3//! QUIC, and for the certificates a user transport returns.

19/// OID for the karyon peer-id custom extension.
20// TODO: not standards-conformant. Replace with a registered one.
21pub(crate) const PEER_ID_EXT_OID: &[u64] = &[0, 0, 0, 0];
22
23const BAD_SIGNATURE_ERR: rustls::Error = InvalidCertificate(CertificateError::BadSignature);
24const BAD_ENCODING_ERR: rustls::Error = InvalidCertificate(CertificateError::BadEncoding);
25
26/// Derive the peer id from a peer's certificate chain post-handshake.
27/// Reuses `verify_cert` so the validation rules stay in one place.
28/// Returns `None` if the chain is empty or the cert lacks our extension.
29pub(crate) fn peer_id_from_certs(certs: &[CertificateDer<'_>]) -> Option<PeerID> {
30    let end_entity = certs.first()?;
31    verify_cert(end_entity).ok()
32}
33
34/// Verifies the given certification.
35pub(crate) fn verify_cert(
36    end_entity: &CertificateDer<'_>,
37) -> std::result::Result<PeerID, rustls::Error> {
38    // Parse the certificate.
39    let cert = parse_cert(end_entity)?;
40
41    // Find our custom extension by OID, not by position.
42    let want_oid = Oid::from(PEER_ID_EXT_OID).map_err(|_| BAD_ENCODING_ERR)?;
43    let ext = cert
44        .extensions()
45        .iter()
46        .find(|e| e.oid == want_oid)
47        .ok_or(BAD_ENCODING_ERR)?;
48
49    // Extract the peer id (public key) and the signature from the extension.
50    let (public_key, signature): (Vec<u8>, Vec<u8>) =
51        yasna::decode_der(ext.value).map_err(|_| BAD_ENCODING_ERR)?;
52
53    // Use the peer id (public key) to verify the extracted signature.
54    let public_key =
55        PublicKey::from_bytes(&KeyPairType::Ed25519, &public_key).map_err(|_| BAD_ENCODING_ERR)?;
56    public_key
57        .verify(cert.public_key().raw, &signature)
58        .map_err(|_| BAD_SIGNATURE_ERR)?;
59
60    // Verify the certificate signature.
61    verify_cert_signature(
62        &cert,
63        cert.tbs_certificate.as_ref(),
64        cert.signature_value.as_ref(),
65    )?;
66
67    PeerID::try_from(public_key).map_err(|_| BAD_ENCODING_ERR)
68}

70/// Parses the given x509 certificate.
71pub(crate) fn parse_cert<'a>(
72    end_entity: &'a CertificateDer<'a>,
73) -> std::result::Result<X509Certificate<'a>, rustls::Error> {
74    let (_, cert) = parse_x509_certificate(end_entity.as_ref()).map_err(|_| BAD_ENCODING_ERR)?;
75
76    if !cert.validity().is_valid() {
77        return Err(InvalidCertificate(CertificateError::NotValidYet));
78    }
79
80    Ok(cert)
81}

83/// Verifies the signature of the given certificate.
84pub(crate) fn verify_cert_signature(
85    cert: &X509Certificate,
86    message: &[u8],
87    signature: &[u8],
88) -> std::result::Result<(), rustls::Error> {
89    let public_key = PublicKey::from_bytes(
90        &KeyPairType::Ed25519,
91        cert.tbs_certificate.subject_pki.subject_public_key.as_ref(),
92    )
93    .map_err(|_| BAD_ENCODING_ERR)?;
94
95    public_key
96        .verify(message, signature)
97        .map_err(|_| BAD_SIGNATURE_ERR)
98}
