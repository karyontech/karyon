14/// A user supplied way to dial and listen for peer connections, set with
15/// `Config::transport`. The node uses it for every endpoint it
16/// `supports`, and the built-in TCP, TLS and QUIC for the others.
17///
18/// The returned streams carry the peer protocol as is. If the transport
19/// authenticates the peer with a karyon certificate, return it from
20/// `ByteStream::peer_certificates` and the handshake checks the peer's id
21/// against it, as with TLS.
22#[async_trait]
23pub trait Transport: Send + Sync {
24    /// Whether this transport handles `endpoint`.
25    fn supports(&self, endpoint: &Endpoint) -> bool;
26
27    /// Opens a connection to `endpoint`.
28    async fn dial(&self, endpoint: &Endpoint) -> Result<Box<dyn ByteStream>>;
29
30    /// Starts listening on `endpoint`.
31    async fn listen(&self, endpoint: &Endpoint) -> Result<Box<dyn TransportListener>>;
32}
33
34/// Accepts the connections of a `Transport` listener.
35#[async_trait]
36pub trait TransportListener: Send + Sync {
37    /// Waits for the next connection. Keep it cheap: it runs in the
38    /// accept loop, so a slow accept holds back the next connections.
39    async fn accept(&self) -> Result<Box<dyn ByteStream>>;
40
41    /// The endpoint the listener is bound to.
42    fn local_endpoint(&self) -> Result<Endpoint>;
43}
