24/// One admitted connection to a peer, from the handshake until it stops:
25/// the wire, its tasks and the slot. The PeerPool runs it, `Sessions`
26/// keeps a clone of it.
27#[derive(Clone)]
28pub(crate) struct PeerSession {
29    pub(crate) info: SessionInfoRef,
30    /// Unique per connection in the pool. A peer that reconnects or whose
31    /// connection is replaced gets a new one, while the id stays the same.
32    pub(crate) conn_id: u64,
33    /// From the handshake, the same on both nodes. Picks which of two
34    /// connections to the same peer stays (see `Sessions::insert`).
35    pub(crate) conn_key: u64,
36    /// Hides the wire shape (single framed pipe vs. per-protocol streams).
37    wire: WireRef,
38    /// Session tasks send the reason the session must stop. `run` waits on
39    /// `stop_rx` and returns it.
40    stop_tx: Sender<Result<()>>,
41    stop_rx: Receiver<Result<()>>,
42    /// The wire and protocol tasks of this session only.
43    task_group: TaskGroupRef,
44    /// The listener or connector slot, freed once the last clone of the
45    /// session is dropped.
46    _slot: Arc<SemaphoreGuard>,
47}
48
49impl PeerSession {
50    /// Builds the session and its wire. Does not wait for the remote.
51    pub(crate) fn new(
52        info: SessionInfoRef,
53        conn: QueuedConn,
54        conn_id: u64,
55        conn_key: u64,
56        slot: SemaphoreGuard,
57        executor: Executor,
58    ) -> Result<Self> {
59        let task_group = Arc::new(TaskGroup::with_executor(executor));
60        let (stop_tx, stop_rx) = async_channel::bounded::<Result<()>>(1);
61        let wire = connection::from_queued(conn, &info.negotiated, &task_group, stop_tx.clone())?;
62
63        Ok(Self {
64            info,
65            conn_id,
66            conn_key,
67            wire,
68            stop_tx,
69            stop_rx,
70            task_group,
71            _slot: Arc::new(slot),
72        })
73    }
74
75    /// Sets up the wire, starts the protocols and waits for the reason to
76    /// stop. The pool stops the session once this returns.
77    pub(crate) async fn run(
78        &self,
79        protocols: Vec<ResolvedProtocol>,
80        broadcaster: Broadcaster,
81        setup_timeout: Duration,
82    ) -> Result<()> {
83        // QUIC opens its protocol streams here. A peer that does not open
84        // them in time is stopped, like on any other connection error.
85        let setup = self.wire.setup(&self.task_group, self.stop_tx.clone());
86        timeout(setup_timeout, setup).await??;
87
88        for (proto_id, constructor) in protocols {
89            trace!("peer {} run protocol {proto_id}", self.info.id);
90            let peer_conn = PeerConn::new(
91                self.info.clone(),
92                self.wire.clone(),
93                proto_id.clone(),
94                broadcaster.clone(),
95            );
96            let protocol = constructor(peer_conn);
97
98            let on_end = {
99                let stop_tx = self.stop_tx.clone();
100                |result: TaskResult<Result<()>>| async move {
101                    if let TaskResult::Completed(res) = result {
102                        if res.is_err() {
103                            error!("protocol {proto_id} stopped");
104                        }
105                        let _ = stop_tx.try_send(res);
106                    }
107                }
108            };
109            self.task_group.spawn_then(protocol.start(), on_end);
110        }
111
112        self.stop_rx.recv().await?
113    }
114
115    pub(crate) async fn send(&self, proto_id: &ProtocolID, msg: Vec<u8>) -> Result<()> {
116        self.wire.send(proto_id, msg).await
117    }
118
119    /// Closes the wire, wakes `run` and cancels the session's tasks.
120    /// Call it once per session.
121    pub(crate) async fn stop(&self) {
122        trace!("peer {} shutting down", self.info.id);
123        let _ = self.wire.shutdown().await;
124        let _ = self.stop_tx.try_send(Ok(()));
125        self.task_group.cancel().await;
126    }
127}
