33/// Connection lifecycle events the Node forwards to discovery.
34#[derive(Debug, Clone)]
35pub enum PeerConnectionEvent {
36    /// A peer joined the pool, inbound or outbound.
37    Connected(PeerID),
38    /// A peer left the pool.
39    Disconnected(PeerID),
40    /// The Node could not dial a peer. The id is `None` for a
41    /// `dial_peers` entry whose id is not known.
42    DialFailed(Option<PeerID>),
43    /// A connection, inbound or outbound, failed the handshake or was
44    /// denied by `AccessControl`. The id is `None` if not verified yet.
45    HandshakeFailed(Option<PeerID>),
46}

48/// Trait that any discovery protocol must implement.
49///
50/// Discovery implementations find peers and yield them via
51/// `recv()`. The Node awaits `recv()` in a loop and handles
52/// connecting - discovery never connects directly. The Node dials
53/// `Config::dial_peers` itself. Only `start` and `recv` are required.
54#[async_trait]
55pub trait Discovery: Send + Sync {
56    /// Start the discovery service. Listen endpoints, if needed by
57    /// the implementation, must be provided at construction time.
58    async fn start(self: Arc<Self>) -> Result<()>;
59
60    /// Shutdown the discovery service.
61    async fn shutdown(&self) {}
62
63    /// Await the next discovered peer. The Node calls this in a
64    /// loop; on shutdown the surrounding task is cancelled.
65    async fn recv(&self) -> DiscoveredPeer;
66
67    /// Receive a peer connection event from the Node.
68    fn on_event(&self, _event: PeerConnectionEvent) {}
69
70    /// Advertise an item (protocol id, swarm key, ...) to other peers.
71    /// `REQUIRED` items are required from peers, `PREFERRED` ones are
72    /// preferred but not required. Other bits are implementation
73    /// defined.
74    fn advertise(&self, _item: &[u8], _flags: ProtocolFlags) {}
75}
