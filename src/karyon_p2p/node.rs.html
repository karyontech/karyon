37/// `Arc` reference to a `Node`.
38pub type NodeRef = Arc<Node>;

40/// Central entry point for the p2p network.
41///
42/// Manages peer connections, discovery, and protocol registration.
43///
44/// # Example
45///
46/// ```no_run
47/// use karyon_core::async_runtime::global_executor;
48/// use karyon_p2p::{Node, Config, keypair::{KeyPair, KeyPairType}};
49///
50/// let key_pair = KeyPair::generate(&KeyPairType::Ed25519);
51/// let config = Config {
52///     listen_endpoints: vec![
53///         "tcp://0.0.0.0:8000".parse().unwrap(),
54///     ],
55///     ..Config::default()
56/// };
57///
58/// let node = Node::new(&key_pair, config, global_executor());
59///
60/// // node.run().await.unwrap();
61/// // node.attach_protocol(MyProto::new).await;
62/// // node.shutdown().await;
63/// ```
64pub struct Node {
65    /// The Configuration for the P2P network.
66    config: ConfigRef,
67
68    /// Identity Key pair
69    key_pair: KeyPair,
70
71    /// Peer ID
72    peer_id: PeerID,
73
74    /// Responsible for network and system monitoring.
75    monitor: MonitorRef,
76
77    /// Discovery instance.
78    discovery: DiscoveryRef,
79
80    /// Attached protocols, shared with the PeerPool.
81    registry: ProtocolRegistryRef,
82
83    /// Owns the sessions and runs their lifecycle.
84    peer_pool: PeerPoolRef,
85
86    /// Where the PeerPool reports peer events.
87    peer_events: PeerEventsRef,
88
89    /// Connector for outbound connections.
90    connector: ConnectorRef<PeerNetMsgCodec>,
91
92    /// Listener for inbound connections.
93    listener: ListenerRef<PeerNetMsgCodec>,
94
95    /// Managing spawned tasks.
96    task_group: TaskGroup,
97}

100    /// Creates a new Node that runs no discovery: it only dials
101    /// `Config::dial_peers`. Use `with_discovery` to find more peers.
102    pub fn new(key_pair: &KeyPair, config: Config, ex: Executor) -> NodeRef {
103        Self::with_discovery(key_pair, config, |_| Arc::new(NoDiscovery), ex)
104    }

106    /// Creates a new Node with a custom discovery. `discovery` builds it
107    /// from what the node shares with it: key pair, PeerID, config,
108    /// monitor and executor.
109    pub fn with_discovery(
110        key_pair: &KeyPair,
111        config: Config,
112        discovery: impl FnOnce(&DiscoveryContext) -> DiscoveryRef,
113        ex: Executor,
114    ) -> NodeRef {
115        let config = Arc::new(config);
116        let monitor = Monitor::new(config.enable_monitor);
117        let peer_id = PeerID::try_from(key_pair.public())
118            .expect("Derive a peer id from the provided key pair.");
119        info!("PeerID: {peer_id}");
120
121        let discovery = discovery(&DiscoveryContext {
122            key_pair: key_pair.clone(),
123            peer_id: peer_id.clone(),
124            config: config.clone(),
125            monitor: monitor.clone(),
126            executor: ex.clone(),
127        });
128
129        // The listener and the connector hand their connections to the
130        // PeerPool over this queue.
131        let conn_queue = AsyncQueue::new(CONN_QUEUE_SIZE);
132        let registry = ProtocolRegistry::new();
133        let peer_events = PeerEvents::new(monitor.clone(), discovery.clone());
134        let admit_config = AdmitConfig {
135            version: config.version.clone(),
136            handshake_timeout: config.handshake_timeout,
137            access_control: config.access_control.clone(),
138        };
139        let peer_pool = PeerPool::new(
140            &peer_id,
141            conn_queue.clone(),
142            registry.clone(),
143            admit_config,
144            peer_events.clone(),
145            ex.clone(),
146        );
147
148        let connector = Connector::new_with_queue(
149            key_pair,
150            &config,
151            conn_queue.clone(),
152            monitor.clone(),
153            ex.clone(),
154        );
155
156        let listener =
157            Listener::new_with_queue(key_pair, &config, conn_queue, monitor.clone(), ex.clone());
158
159        let task_group = TaskGroup::with_executor(ex);
160
161        Arc::new(Self {
162            key_pair: key_pair.clone(),
163            peer_id,
164            monitor,
165            discovery,
166            config,
167            registry,
168            peer_pool,
169            peer_events,
170            connector,
171            listener,
172            task_group,
173        })
174    }

176    /// Run the Node, starting listeners, PeerPool, and Discovery.
177    pub async fn run(self: &Arc<Self>) -> Result<()> {
178        // Core protocols (PING) are attached before the pool starts so
179        // they're advertised on every handshake.
180        self.attach_core_protocols().await?;
181
182        self.peer_pool.start().await?;
183
184        // Start data listeners.
185        for endpoint in &self.config.listen_endpoints {
186            let resolved = self.listener.start(endpoint.clone()).await?;
187            info!("Listening on {resolved}");
188        }
189
190        // Start discovery.
191        self.discovery.clone().start().await?;
192
193        // Spawn task to connect discovered peers.
194        self.task_group.spawn_then(
195            {
196                let this = self.clone();
197                async move { this.connect_discovered_peers().await }
198            },
199            |res: TaskResult<()>| async move {
200                debug!("connect_discovered_peers task ended: {res}");
201            },
202        );
203
204        Ok(())
205    }
206
207    /// Dial `Config::dial_peers` once, then the peers the discovery
208    /// yields. Runs forever; the task_group cancels it on Node::shutdown.
209    async fn connect_discovered_peers(self: Arc<Self>) {
210        for endpoint in self.config.dial_peers.iter() {
211            let peer = DiscoveredPeer {
212                peer_id: None,
213                addrs: PeerAddr::from_endpoint(endpoint, 0).into_iter().collect(),
214            };
215            if self.dial(peer).await.is_err() {
216                return;
217            }
218        }
219
220        loop {
221            let peer = self.discovery.recv().await;
222            if self.dial(peer).await.is_err() {
223                return;
224            }
225        }
226    }
227
228    /// Dial `peer` in its own task once an outbound slot is free. Skips
229    /// peers with no usable address, denied or already connected. Fails
230    /// only if the outbound slots are closed.
231    async fn dial(self: &Arc<Self>, peer: DiscoveredPeer) -> Result<()> {
232        let supported = [Protocol::Tcp, Protocol::Tls, Protocol::Quic];
233        let Some(endpoint) = pick_endpoint(&peer.addrs, &supported, self.config.transport.as_ref())
234        else {
235            return Ok(());
236        };
237
238        if !self.config.access_control.allow(
239            &Subject::Endpoint(&endpoint),
240            Action::Connect(ConnDirection::Outbound),
241        ) {
242            debug!("Skipped dialing denied endpoint {endpoint}");
243            return Ok(());
244        }
245
246        // Already connected: a second connection would only be
247        // rejected or replace the first.
248        if let Some(pid) = &peer.peer_id
249            && self.peer_pool.contains_peer(pid).await
250        {
251            return Ok(());
252        }
253
254        // Take the slot here, so the caller waits when all slots are
255        // busy. The dial runs in its own task, so one slow endpoint
256        // does not hold back the next dial.
257        let slot = self
258            .connector
259            .acquire_slot()
260            .await
261            .inspect_err(|err| error!("Acquire outbound slot: {err}"))?;
262
263        let peer_id = peer.peer_id;
264        let this = self.clone();
265        self.task_group.spawn(async move {
266            let res = this
267                .connector
268                .connect_and_queue(&endpoint, &peer_id, slot)
269                .await;
270            if res.is_err() {
271                this.monitor
272                    .notify(PoolEvent::ConnectFailed(peer_id.clone(), endpoint))
273                    .await;
274                this.discovery
275                    .on_event(PeerConnectionEvent::DialFailed(peer_id));
276            }
277        });
278        Ok(())
279    }

281    /// Attach a custom protocol. karyon runs the constructor closure
282    /// once per connected peer with a typed `PeerConn` scoped to this
283    /// protocol. The protocol type is inferred from the closure's
284    /// return value. The protocol id is advertised through discovery
285    /// according to `P::flags()`.
286    ///
287    /// ```ignore
288    /// node.attach_protocol(MyProtocol::new).await?;
289    /// ```
290    pub async fn attach_protocol<P: ProtocolTrait + 'static>(
291        &self,
292        c: impl Fn(PeerConn) -> P + Send + Sync + 'static,
293    ) -> Result<()> {
294        let meta = ProtocolMeta {
295            version: P::version()?,
296            flags: P::flags(),
297        };
298        let c = move |conn| Arc::new(c(conn)) as ProtocolRef;
299        self.registry.insert(P::id(), Arc::new(c), meta).await;
300        self.advertise(P::id().as_bytes(), P::flags());
301        Ok(())
302    }
303
304    /// Attach the core protocols (PING). Called once during `run`.
305    async fn attach_core_protocols(&self) -> Result<()> {
306        let interval = self.config.ping_interval;
307        let timeout = self.config.ping_timeout;
308        self.attach_protocol(move |peer| PingProtocol::new(peer, interval, timeout))
309            .await
310    }

312    /// Advertise an item (protocol id, swarm key, ...) through

393    /// Shuts down the Node.
394    pub async fn shutdown(&self) {
395        self.discovery.shutdown().await;
396        // Stop everything that hands connections to the PeerPool first
397        // (the dial tasks run in this task group), so the pool can drop
398        // the rest of its queue.
399        self.listener.shutdown().await;
400        self.connector.shutdown().await;
401        self.task_group.cancel().await;
402        self.peer_pool.shutdown().await;
403    }
