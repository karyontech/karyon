34/// Internal dial result. Generic over the codec used to frame the
35/// resulting connection. Carries the optional cert-derived PeerID so
36/// the caller can stamp it on the QueuedConn for the application
37/// handshake to enforce.
38enum DialResult<C: Codec<ByteBuffer> + Default + Clone> {
39    /// TCP or TLS: a framed connection.
40    Channel(FramedConn<C>, Option<PeerID>),
41    /// QUIC: handshake stream + full QUIC connection.
42    #[cfg(feature = "quic")]
43    Quic(FramedConn<C>, quic::QuicConn, Option<PeerID>),
44}
45
46/// `Arc` reference to a `Connector`.
47pub type ConnectorRef<C> = Arc<Connector<C>>;
48
49/// Creates outbound connections with other peers. Generic over the
50/// codec applied to the resulting framed stream so the same machinery
51/// (TLS, retries, slots, monitor events) serves both the peer
52/// data-plane (`PeerNetMsgCodec`) and a discovery's own messages.
53pub struct Connector<C: Codec<ByteBuffer> + Default + Clone> {
54    key_pair: KeyPair,
55    task_group: TaskGroup,
56    connection_slots: Semaphore,
57    max_retries: usize,
58    /// Max time for one dial: connect plus the TLS or QUIC handshake.
59    dial_timeout: Duration,
60    /// Hands dialed connections to the PeerPool. Only on the Node's own
61    /// connector.
62    conn_queue: Option<ConnQueueRef>,
63    /// Dials the endpoints it supports instead of the built-in transports.
64    transport: Option<TransportRef>,
65    monitor: MonitorRef,
66    _codec: PhantomData<C>,
67}

122    /// Connect and return the framed connection. The slot is held until
123    /// the returned guard is dropped.
124    pub async fn connect(
125        &self,
126        endpoint: &Endpoint,
127        peer_id: &Option<PeerID>,
128    ) -> Result<(FramedConn<C>, SemaphoreGuard)> {
129        let slot = self.acquire_slot().await?;
130        let result = self.connect_internal(endpoint, peer_id).await?;
131        match result {
132            DialResult::Channel(conn, _) => Ok((conn, slot)),
133            #[cfg(feature = "quic")]
134            DialResult::Quic(conn, _, _) => Ok((conn, slot)),
135        }
136    }
137
138    /// Wait for a free connection slot.
139    pub(crate) async fn acquire_slot(&self) -> Result<SemaphoreGuard> {
140        Ok(self.connection_slots.acquire().await?)
141    }
142
143    /// Dial with retries. Each try is bounded by `dial_timeout`, so an
144    /// endpoint that accepts and then stays silent can not hang the dial.
145    async fn connect_internal(
146        &self,
147        endpoint: &Endpoint,
148        peer_id: &Option<PeerID>,
149    ) -> Result<DialResult<C>> {
150        let mut retry = 0;
151        let backoff = Backoff::new(500, 2000);
152        while retry < self.max_retries {
153            let dial = async { timeout(self.dial_timeout, self.dial(endpoint, peer_id)).await? };
154            match dial.await {
155                Ok(result) => {
156                    self.monitor
157                        .notify(ConnectionKind::Connected(endpoint.clone()))
158                        .await;
159                    return Ok(result);
160                }
161                Err(err) => {
162                    error!("Failed to connect to {endpoint}: {err}");
163                }
164            }
165
166            self.monitor
167                .notify(ConnectionKind::ConnectRetried(endpoint.clone()))
168                .await;
169
170            backoff.sleep().await;
171            warn!("try to reconnect {endpoint}");
172            retry += 1;
173        }
174
175        self.monitor
176            .notify(ConnectionKind::ConnectFailed(endpoint.clone()))
177            .await;
178
179        Err(Error::Timeout)
180    }
181
182    /// Dial, selecting transport based on endpoint type.
183    async fn dial(&self, endpoint: &Endpoint, peer_id: &Option<PeerID>) -> Result<DialResult<C>> {
184        if let Some(transport) = self.transport.as_ref().filter(|t| t.supports(endpoint)) {
185            let stream = transport.dial(endpoint).await?;
186            let vpid = stream
187                .peer_certificates()
188                .as_deref()
189                .and_then(peer_id_from_certs);
190            return Ok(DialResult::Channel(framed(stream, C::default()), vpid));
191        }
192
193        match endpoint {
194            Endpoint::Tcp(..) => {
195                let stream = tcp::connect(endpoint, Default::default()).await?;
196                let conn = framed(stream, C::default());
197                Ok(DialResult::Channel(conn, None))
198            }
199            Endpoint::Tls(..) => {
200                let tls_config = karyon_net::tls::ClientTlsConfig {
201                    client_config: tls_client_config(&self.key_pair, peer_id.clone())?,
202                    dns_name: DNS_NAME.to_string(),
203                };
204                let stream = tcp::connect(endpoint, Default::default()).await?;
205                let tls_layer = karyon_net::tls::TlsLayer::client(tls_config);
206                let tls_stream = ClientLayer::handshake(&tls_layer, stream).await?;
207                // Extract the peer cert before framing consumes the stream.
208                let vpid = tls_stream
209                    .peer_certificates()
210                    .as_deref()
211                    .and_then(peer_id_from_certs);
212                let conn = framed(tls_stream, C::default());
213                Ok(DialResult::Channel(conn, vpid))
214            }
215            #[cfg(feature = "quic")]
216            Endpoint::Quic(..) => {
217                let rustls_config = quic_client_config(&self.key_pair, peer_id.clone())?;
218                let client_config = quic::ClientQuicConfig::from_rustls(rustls_config, DNS_NAME);
219                let quic_conn = quic::QuicEndpoint::dial(endpoint, client_config).await?;
220
221                let vpid = quic_conn
222                    .peer_certificates()
223                    .as_deref()
224                    .and_then(peer_id_from_certs);
225
226                // First stream for handshake via StreamMux.
227                let stream = quic_conn.open_stream().await?;
228                let conn = framed(stream, C::default());
229
230                Ok(DialResult::Quic(conn, quic_conn, vpid))
231            }
232            _ => Err(Error::UnsupportedEndpoint(endpoint.to_string())),
233        }
234    }
235}

237// `connect_and_queue` only lives on the peer-plane Connector since the
238// PeerPool takes only peer-plane connections.
239impl Connector<PeerNetMsgCodec> {
240    /// Create a new Connector that hands its connections to the PeerPool.
241    pub(crate) fn new_with_queue(
242        key_pair: &KeyPair,
243        config: &Config,
244        conn_queue: ConnQueueRef,
245        monitor: MonitorRef,
246        ex: Executor,
247    ) -> ConnectorRef<PeerNetMsgCodec> {
248        Arc::new(Self {
249            key_pair: key_pair.clone(),
250            max_retries: config.max_connect_retries,
251            dial_timeout: Duration::from_secs(config.handshake_timeout),
252            task_group: TaskGroup::with_executor(ex),
253            monitor,
254            connection_slots: Semaphore::new(config.outbound_slots),
255            conn_queue: Some(conn_queue),
256            transport: config.transport.clone(),
257            _codec: PhantomData,
258        })
259    }
260
261    /// Connect and hand the connection to the PeerPool, which handshakes
262    /// it. Handshake failure and disconnect are observable via
263    /// `Node::register_peer_events`, not via this call. `slot` comes
264    /// from `acquire_slot` and is held until the connection is closed.
265    pub(crate) async fn connect_and_queue(
266        &self,
267        endpoint: &Endpoint,
268        peer_id: &Option<PeerID>,
269        slot: SemaphoreGuard,
270    ) -> Result<()> {
271        let conn_queue = self.conn_queue.as_ref().ok_or_else(|| {
272            Error::Config("connect_and_queue called on Connector built without a PeerPool".into())
273        })?;
274
275        let conn = match self.connect_internal(endpoint, peer_id).await? {
276            DialResult::Channel(conn, vpid) => QueuedConn::new(conn, ConnDirection::Outbound, vpid),
277            #[cfg(feature = "quic")]
278            DialResult::Quic(conn, quic_conn, vpid) => {
279                QueuedConn::new(conn, ConnDirection::Outbound, vpid).map(|c| c.with_quic(quic_conn))
280            }
281        };
282
283        match conn {
284            Ok(conn) => {
285                conn_queue.push(Incoming { conn, slot }).await;
286                Ok(())
287            }
288            Err(err) => {
289                self.monitor
290                    .notify(ConnectionKind::Disconnected(endpoint.clone()))
291                    .await;
292                Err(err)
293            }
294        }
295    }
