31/// Internal dial result: the stream, not framed yet, and the peer id its
32/// certificate proves, which the handshake then enforces.
33pub(crate) enum DialResult {
34    /// TCP, TLS or a user transport: one stream.
35    Channel(Box<dyn ByteStream>, Option<PeerID>),
36    /// QUIC: the handshake stream and the QUIC connection.
37    #[cfg(feature = "quic")]
38    Quic(Box<dyn ByteStream>, QuicConn, Option<PeerID>),
39}

41/// `Arc` reference to a `Connector`.
42pub type ConnectorRef<C> = Arc<Connector<C>>;
43
44/// Creates outbound connections with other peers. Generic over the
45/// codec applied to the resulting framed stream so the same machinery
46/// (TLS, retries, slots, monitor events) serves both the peer
47/// data-plane (`PeerNetMsgCodec`) and a discovery's own messages.
48pub struct Connector<C: Codec<ByteBuffer> + Default + Clone> {
49    /// Dials the endpoints `transport` does not support.
50    builtin: BuiltinTransports,
51    task_group: TaskGroup,
52    connection_slots: Semaphore,
53    max_retries: usize,
54    /// Max time for one dial: connect plus the TLS or QUIC handshake.
55    dial_timeout: Duration,
56    /// Hands dialed connections to the PeerPool. Only on the Node's own
57    /// connector.
58    conn_queue: Option<ConnQueueRef>,
59    /// Dials the endpoints it supports instead of the built-in transports.
60    transport: Option<TransportRef>,
61    monitor: MonitorRef,
62    _codec: PhantomData<C>,
63}

65impl<C> Connector<C>
66where
67    C: Codec<ByteBuffer, Error = karyon_net::Error> + Default + Clone + Send + Sync + 'static,
68{
69    /// Create a Connector for a discovery's own connections. The key
70    /// pair, monitor, executor and the dial timeout (`handshake_timeout`),
71    /// transport and QUIC runtime of the node's config come from `ctx`.
72    pub fn new(ctx: &DiscoveryContext, max_retries: usize, slots: usize) -> ConnectorRef<C> {
73        Self::build(
74            &ctx.key_pair,
75            &ctx.config,
76            max_retries,
77            slots,
78            None,
79            ctx.monitor.clone(),
80            ctx.executor.clone(),
81        )
82    }
83
84    /// Builds a Connector. `conn_queue` is set only on the Node's own.
85    fn build(
86        key_pair: &KeyPair,
87        config: &Config,
88        max_retries: usize,
89        slots: usize,
90        conn_queue: Option<ConnQueueRef>,
91        monitor: MonitorRef,
92        ex: Executor,
93    ) -> ConnectorRef<C> {
94        Arc::new(Self {
95            builtin: BuiltinTransports::new(key_pair, config),
96            max_retries,
97            dial_timeout: Duration::from_secs(config.handshake_timeout),
98            task_group: TaskGroup::with_executor(ex),
99            monitor,
100            connection_slots: Semaphore::new(slots),
101            conn_queue,
102            transport: config.transport.clone(),
103            _codec: PhantomData,
104        })
105    }
106
107    pub async fn shutdown(&self) {
108        self.task_group.cancel().await;
109    }
110
111    /// Connect and return the framed connection. The slot is held until
112    /// the returned guard is dropped.
113    pub async fn connect(
114        &self,
115        endpoint: &Endpoint,
116        peer_id: &Option<PeerID>,
117    ) -> Result<(FramedConn<C>, SemaphoreGuard)> {
118        let slot = self.acquire_slot().await?;
119        let stream = match self.connect_internal(endpoint, peer_id).await? {
120            DialResult::Channel(stream, _) => stream,
121            #[cfg(feature = "quic")]
122            DialResult::Quic(stream, _, _) => stream,
123        };
124        Ok((framed(stream, C::default()), slot))
125    }
126
127    /// Wait for a free connection slot.
128    pub(crate) async fn acquire_slot(&self) -> Result<SemaphoreGuard> {
129        Ok(self.connection_slots.acquire().await?)
130    }
131
132    /// Dial with retries. Each try is bounded by `dial_timeout`, so an
133    /// endpoint that accepts and then stays silent can not hang the dial.
134    async fn connect_internal(
135        &self,
136        endpoint: &Endpoint,
137        peer_id: &Option<PeerID>,
138    ) -> Result<DialResult> {
139        let mut retry = 0;
140        let ex = self.task_group.executor();
141        let backoff = Backoff::with_executor(500, 2000, ex.clone());
142        while retry < self.max_retries {
143            let dial = async {
144                ex.timeout(self.dial_timeout, self.dial(endpoint, peer_id))
145                    .await?
146            };
147            match dial.await {
148                Ok(result) => {
149                    self.monitor
150                        .notify(ConnectionKind::Connected(endpoint.clone()))
151                        .await;
152                    return Ok(result);
153                }
154                Err(err) => {
155                    error!("Failed to connect to {endpoint}: {err}");
156                }
157            }
158
159            self.monitor
160                .notify(ConnectionKind::ConnectRetried(endpoint.clone()))
161                .await;
162
163            backoff.sleep().await;
164            warn!("try to reconnect {endpoint}");
165            retry += 1;
166        }
167
168        self.monitor
169            .notify(ConnectionKind::ConnectFailed(endpoint.clone()))
170            .await;
171
172        Err(Error::Timeout)
173    }
174
175    /// Dial with the user transport if it supports `endpoint`, else with
176    /// the built-in ones.
177    async fn dial(&self, endpoint: &Endpoint, peer_id: &Option<PeerID>) -> Result<DialResult> {
178        if let Some(transport) = self.transport.as_ref().filter(|t| t.supports(endpoint)) {
179            let stream = transport.dial(endpoint).await?;
180            let vpid = stream
181                .peer_certificates()
182                .as_deref()
183                .and_then(peer_id_from_certs);
184            return Ok(DialResult::Channel(stream, vpid));
185        }
186
187        #[cfg(feature = "quic")]
188        if endpoint.is_quic() {
189            let ex = self.task_group.executor();
190            return self.builtin.dial_quic(endpoint, peer_id, ex).await;
191        }
192
193        self.builtin.dial(endpoint, peer_id).await
194    }
195}

197// `connect_and_queue` only lives on the peer-plane Connector since the
198// PeerPool takes only peer-plane connections.
199impl Connector<PeerNetMsgCodec> {
200    /// Create a new Connector that hands its connections to the PeerPool.
201    pub(crate) fn new_with_queue(
202        key_pair: &KeyPair,
203        config: &Config,
204        conn_queue: ConnQueueRef,
205        monitor: MonitorRef,
206        ex: Executor,
207    ) -> ConnectorRef<PeerNetMsgCodec> {
208        Self::build(
209            key_pair,
210            config,
211            config.max_connect_retries,
212            config.outbound_slots,
213            Some(conn_queue),
214            monitor,
215            ex,
216        )
217    }
218
219    /// Connect and hand the connection to the PeerPool, which handshakes
220    /// it. Handshake failure and disconnect are observable via
221    /// `Node::register_peer_events`, not via this call. `slot` comes
222    /// from `acquire_slot` and is held until the connection is closed.
223    pub(crate) async fn connect_and_queue(
224        &self,
225        endpoint: &Endpoint,
226        peer_id: &Option<PeerID>,
227        slot: SemaphoreGuard,
228    ) -> Result<()> {
229        let conn_queue = self.conn_queue.as_ref().ok_or_else(|| {
230            Error::Config("connect_and_queue called on Connector built without a PeerPool".into())
231        })?;
232
233        let conn = match self.connect_internal(endpoint, peer_id).await? {
234            DialResult::Channel(stream, vpid) => QueuedConn::new(
235                framed(stream, PeerNetMsgCodec::new()),
236                ConnDirection::Outbound,
237                vpid,
238            ),
239            #[cfg(feature = "quic")]
240            DialResult::Quic(stream, quic_conn, vpid) => {
241                let conn = framed(stream, PeerNetMsgCodec::new());
242                QueuedConn::new(conn, ConnDirection::Outbound, vpid).map(|c| c.with_quic(quic_conn))
243            }
244        };
245
246        match conn.map(|c| c.with_dialed(peer_id.clone())) {
247            Ok(conn) => {
248                conn_queue.push(Incoming { conn, slot }).await;
249                Ok(())
250            }
251            Err(err) => {
252                self.monitor
253                    .notify(ConnectionKind::Disconnected(endpoint.clone()))
254                    .await;
255                Err(err)
256            }
257        }
258    }
259}
