1use crate::{
2    Error, Result,
3    protocol::{Broadcaster, ProtocolEvent, ProtocolID},
4    session::{SessionInfoRef, WireRef},
5};
6
7/// Per-protocol view of a peer with the protocol id baked in. Built
8/// by karyon and handed to the protocol constructor. A protocol can
9/// only send and receive on its own id.
10pub struct PeerConn {
11    info: SessionInfoRef,
12    wire: WireRef,
13    proto_id: ProtocolID,
14    broadcaster: Broadcaster,
15}
16
17impl PeerConn {
18    /// karyon-internal constructor. User code receives a fully-built
19    /// `PeerConn` from the protocol constructor closure.
20    pub(crate) fn new(
21        info: SessionInfoRef,
22        wire: WireRef,
23        proto_id: ProtocolID,
24        broadcaster: Broadcaster,
25    ) -> Self {
26        Self {
27            info,
28            wire,
29            proto_id,
30            broadcaster,
31        }
32    }
33
34    /// Send pre-encoded payload bytes to the peer on this protocol.
35    pub async fn send(&self, payload: Vec<u8>) -> Result<()> {
36        self.wire.send(&self.proto_id, payload).await
37    }
38
39    /// Receive the next message bytes. Returns `Err(PeerShutdown)`
40    /// when the peer is closing -- callers can treat that as an
41    /// orderly end of stream.
42    pub async fn recv(&self) -> Result<Vec<u8>> {
43        match self.wire.recv(&self.proto_id).await? {
44            ProtocolEvent::Message(bytes) => Ok(bytes),
45            ProtocolEvent::Shutdown => Err(Error::PeerShutdown),
46        }
47    }
48
49    /// Broadcast pre-encoded bytes to every connected peer on this
50    /// protocol, this one included. Does nothing once the node is shut
51    /// down.
52    pub async fn broadcast(&self, payload: Vec<u8>) {
53        self.broadcaster.broadcast(&self.proto_id, payload).await
54    }
55
56    /// What is known about this session with the remote peer. Cheap to
57    /// clone.
58    pub fn info(&self) -> &SessionInfoRef {
59        &self.info
60    }
61}
